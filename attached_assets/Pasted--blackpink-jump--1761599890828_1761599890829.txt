2) 곡 식별 해시 + 시드 충돌 제거 (utils.js)
export function xorshift32(seed){ let x=seed|0||1; return ()=>((x^=x<<13,x^=x>>>17,x^=x<<5)>>>0)/4294967296; }

export async function hash32(str){
  const b=new TextEncoder().encode(str);
  const d=await crypto.subtle.digest('SHA-256', b);
  const v=new DataView(d);
  let h=0; for(let i=0;i<v.byteLength;i+=4) h^=v.getUint32(i);
  return h>>>0;
}

export async function audioHash32(audioBuffer, seconds=90){
  const sr=audioBuffer.sampleRate, frames=Math.min(audioBuffer.length, seconds*sr|0);
  const chs=audioBuffer.numberOfChannels, mono=new Float32Array(frames);
  for(let c=0;c<chs;c++){ const cd=audioBuffer.getChannelData(c); for(let i=0;i<frames;i++) mono[i]+=cd[i]/chs; }
  const factor=Math.max(1, Math.floor(sr/11025)), len=Math.ceil(frames/factor), ds=new Float32Array(len);
  for(let i=0,j=0;i<frames;i+=factor) ds[j++]=mono[i];
  const raw=new Uint8Array(ds.length*2);
  for(let i=0;i<ds.length;i++){ const q=((Math.max(-1,Math.min(1,ds[i]))*32767)|0); raw[i*2]=q&255; raw[i*2+1]=(q>>8)&255; }
  const d=await crypto.subtle.digest('SHA-256', raw);
  const v=new DataView(d); let h=0; for(let i=0;i<v.byteLength;i+=4) h^=v.getUint32(i); return h>>>0;
}

export function songKey(meta, aHash){ return `${meta.path||meta.title||'untitled'}@${aHash}`; }
